using Goo;
using System.Diagnostics;
using System.Text.Json;

internal static class Program
{
    internal static Container Scene(bool wide, bool motion = true)
    {
        var root = new Container {
            Width = 300, Height = 260, Padding = 10, Gap = 10,
            FlexDirection = FlexDirection.Row, FlexWrap = FlexWrap.Wrap,
            AlignContent = AlignContent.FlexStart,
            BackgroundColor = Color.Rgb(20, 27, 40),
            OutlineColor = Color.Rgb(255, 100, 100), OutlineWidth = 2, OutlineOffset = -10
        };
        root.Children.Add(new Container {
            Key = "first", Width = 130, Height = 50, FlexShrink = 0,
            BorderRadius = 12, BackgroundColor = Color.Rgb(62, 78, 102)
        });
        var card = new Container {
            Key = "card", Width = wide ? 260 : 140, Height = wide ? 100 : 50,
            FlexShrink = 0, Padding = 8, BorderWidth = 2, BorderRadius = 20,
            BorderColor = Color.Rgb(85, 185, 245), BackgroundColor = Color.Rgb(35, 56, 76),
            Overflow = Overflow.Hidden,
            LayoutTransition = motion ? new LayoutTransition(100, Easing.Linear) : null
        };
        card.Children.Add(new Text {
            Content = "Wrapping card", FontFamily = "LayoutBoundsProbe", FontSize = 16,
            Height = 24, Color = Color.White
        });
        root.Children.Add(card);
        return root;
    }

    private static Container Grid(bool wide, bool motion)
    {
        var root = new Container {
            Width = Length.Percent(100), Height = Length.Percent(100), Padding = 16, Gap = 8,
            FlexDirection = FlexDirection.Row, FlexWrap = FlexWrap.Wrap,
            AlignContent = AlignContent.FlexStart
        };
        for (var i = 0; i < 128; i++)
            root.Children.Add(new Container {
                Key = $"card-{i}", Width = wide ? 116 : 54, Height = wide ? 56 : 48,
                FlexShrink = 0, BorderRadius = 12, BorderWidth = 2,
                BorderColor = Color.Rgb(85, 185, 245), BackgroundColor = Color.Rgb(35, 56, 76),
                LayoutTransition = motion ? new LayoutTransition(100, Easing.Linear) : null
            });
        return root;
    }

    private static object Measure(bool active)
    {
        var pump = new MotionPump();
        var rec = new Reconciler { Res = new Resolver(), Pump = pump };
        var root = rec.Mount(Grid(false, active));
        var layout = new Layout();
        var compiler = new VulkanSceneCompiler(2048);
        layout.Calculate(root, 1024, 1024);
        compiler.Compile(root, Color.Black, 1024, 1024);
        var layoutTicks = new long[1024];
        var paintTicks = new long[1024];
        long allocated = 0;
        try
        {
            for (var frame = -256; frame < 1024; frame++)
            {
                var index = frame + 256;
                if (active && index % 64 == 0)
                {
                    rec.Diff(root, Grid(index % 128 == 0, true));
                    layout.MarkStructureDirty();
                    layout.Calculate(root, 1024, 1024);
                }
                var bytes = GC.GetAllocatedBytesForCurrentThread();
                var started = Stopwatch.GetTimestamp();
                if (active) { pump.Sweep(0.0015); layout.RefreshRects(root); }
                else layout.Calculate(root, 1024 + index % 2 * 16, 1024);
                var laidOut = Stopwatch.GetTimestamp();
                if (frame >= 0) allocated += GC.GetAllocatedBytesForCurrentThread() - bytes;
                compiler.Compile(root, Color.Black, 1024, 1024);
                var painted = Stopwatch.GetTimestamp();
                if (frame >= 0) { layoutTicks[frame] = laidOut - started; paintTicks[frame] = painted - laidOut; }
            }
            Array.Sort(layoutTicks);
            Array.Sort(paintTicks);
            double Us(long ticks) => ticks * 1e6 / Stopwatch.Frequency;
            return new {
                active, nodes = 129, frames = 1024,
                layoutP50Us = Us(layoutTicks[512]), layoutP99Us = Us(layoutTicks[1013]),
                paintP50Us = Us(paintTicks[512]), paintP99Us = Us(paintTicks[1013]),
                layoutAllocatedBytesPerFrame = allocated / 1024.0
            };
        }
        finally { NodeLifecycle.DisposeTree(root); }
    }

    private static void Render(string output)
    {
        Directory.CreateDirectory(output);
        using var font = FontSource.FromPath("LayoutBoundsProbe", 400, false,
            "/work/tests/Goo.VulkanProof/Assets/Text/VendSans-VariableFont_wght.ttf");
        font.Register();
        var cell = new ProbeCell();
        var window = new Window { Root = cell, Size = new LayoutSize(300, 260),
            Title = "Layout transition bounds verification", VSync = false };
        try
        {
            window.Open();
            WindowReadbackTestFixture.ForceRender(window, 0);
            Capture("initial");
            cell.Wide = true;
            cell.Rebuild();
            WindowReadbackTestFixture.ForceRender(window, 0);
            Capture("000");
            for (var frame = 1; frame <= 4; frame++)
            {
                WindowReadbackTestFixture.ForceRender(window, 0.025);
                Capture((frame * 25).ToString("000"));
            }
            void Capture(string name)
            {
                WindowReadbackTestFixture.DrainWindowQueue(window, 10000);
                var status = WindowReadbackTestFixture.Request(window, 300, 260);
                if (status == WindowReadbackRequestStatus.NotReady)
                {
                    WindowReadbackTestFixture.DrainWindowQueue(window, 10000);
                    status = WindowReadbackTestFixture.Request(window, 300, 260);
                }
                if (status != WindowReadbackRequestStatus.Accepted)
                    throw new InvalidOperationException("Readback was not accepted: " + status);
                var deadline = Stopwatch.StartNew();
                VulkanReadbackResult? frame = null;
                while (frame == null && deadline.Elapsed.TotalSeconds < 10)
                {
                    WindowReadbackTestFixture.Poll(window);
                    frame = WindowReadbackTestFixture.Take(window);
                    if (frame == null) Thread.Sleep(1);
                }
                if (frame == null) throw new TimeoutException("Frame readback did not complete");
                File.WriteAllBytes(Path.Combine(output, name + ".png"),
                    ImageSource.EncodePng((int)frame.Width, (int)frame.Height, frame.Pixels, true, int.MaxValue));
                var card = window.Tree!.Children[1];
                var text = card.Children[0];
                File.WriteAllText(Path.Combine(output, name + ".json"), JsonSerializer.Serialize(new {
                    name, x = card.Rect.X, y = card.Rect.Y, width = card.Rect.W, height = card.Rect.H,
                    textX = text.Rect.X, textY = text.Rect.Y, textWidth = text.Rect.W, textHeight = text.Rect.H,
                    matrix = new { a = TransformGeometry.WindowMatrix(text).A, d = TransformGeometry.WindowMatrix(text).D }
                }));
            }
        }
        finally { window.Close(); }
    }

    private static void Main(string[] args)
    {
        var includesSize = typeof(LayoutTransitionState).GetField("lastWidth",
            System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic) != null;
        Console.WriteLine("Loaded bounds interpolation: " + includesSize);
        if (args.Length > 2 && includesSize != (args[2] == "feature"))
            throw new InvalidOperationException("Loaded assembly does not match the requested source revision");
        if (args[0] == "render") { Render(args[1]); return; }
        var rows = new List<object>();
        for (var run = 0; run < 7; run++)
        {
            rows.Add(new { run, sample = Measure(false) });
            rows.Add(new { run, sample = Measure(true) });
        }
        File.WriteAllText(args[1], JsonSerializer.Serialize(rows, new JsonSerializerOptions { WriteIndented = true }));
    }
}
